# Dependencies for the ingest scripts.
# Install all with: pip install -r requirements.txt
#
# GitHub Actions installs these in both the scan-tracked-channels and
# process-queue workflows. The article script degrades gracefully if
# readability-lxml / html2text are missing; the YouTube script requires
# youtube-transcript-api.

# Required for YouTube ingestion: fetches transcripts via the public
# transcript endpoint without the bot-detection issues that plague
# yt-dlp on GitHub Actions runners and other datacenter IPs.
youtube-transcript-api

# Required for --from-queue and --from-tracked modes
pyyaml

# Improves article ingestion quality (ingest-article.py falls back to stdlib if absent)
readability-lxml
html2text
